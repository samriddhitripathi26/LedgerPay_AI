Code Needed for this Integration:

<!-- 1. Include the LedgerPay SDK -->
<script src="http://localhost:4000/sdk/ledgerpay.js"></script>

<!-- 2. Trigger Checkout with Custom Options -->
<script>
  const lp = new LedgerPay({ publishableKey: 'pk_live_acme_corp' });
  
  document.getElementById('pay-btn').onclick = () => {
    lp.openCheckout({
      amount: 24999,
      currency: 'INR',
      orderId: 'ORD_ACME_9941',
      customer: { email: 'billing@enterprise.com', name: 'Rajesh Nair' },
      onSuccess: (tx) => alert('Payment completed: ' + tx.transactionId)
    });
  };
</script>
